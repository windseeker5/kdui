{# Lightweight, server-rendered bar chart component. Default 160. bar_color {string} CSS color for bars. Uses var(--chart-1) by default. show_labels {bool} Render X-axis labels. Default True. class_ {string} Extra classes for the .chart wrapper. attrs {dict} Additional attributes for the .chart wrapper. #} {% macro chart(data=[], label_key='label', value_key='value', height=160, bar_color='var(--chart-1)', show_labels=true, class_='', attrs={}) %} {% set values = data | map(attribute=value_key) | list %} {% set max_value = [values | max, 1] | max if values else 1 %} {% set count = data | length %} {% set padding = 24 %} {% set bottom = 28 if show_labels else 8 %} {% set chart_h = height - bottom %} {% set gap = 12 %} {% set bar_w = (400 - (count + 1) * gap) / count if count > 0 else 0 %}
{% endmacro %}
